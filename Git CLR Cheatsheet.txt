
####### Github Commands ######

#Pull latest from Main apply changes to local
#Use this when pulling data-services repo
git pull --rebase

#list out branches
git branch

#add new branch
git branch NEWBRANCHNAME

#switch to use that branch
git checkout BRANCHNAME

#create and checkout branch
git checkout -b BRANCHNAME

#push branch into remote repository
git push --set-upstream origin BRANCHNAME
git push --set-upstream REMOTEREPO BRANCHNAME

#add files to Staging area of branch
git add FILENAME
git status 

#locally commits files added to Branch
git commit -m "SubjectOfCommit"

#send locally commited changes to Git server online
git push -u origin BRANCHNAME
git push -u REMOTEREPO BRANCHNAME

#get list of remote repos
git remote show
git remote -v

#DO Pull Requests and Commit in GITHUB

#After Commited switch back to master branch, delete previous branch, and do a PULL to refresh
#DELETES a branch
git branch -d BRANCHNAME

#Pull down master branch
git pull


#pull the master while in a branch
git pull origin master

#merge with message 
git merge origin -m "MESSAGE"

#Refresh Master and Merge Master to Branch
git checkout master
git pull
git checkout BRANCH_NAME
git merge master
#OPTIONAL
ls
git reflog
#to push Branch back onto Mastergit push origin BRANCH_NAME

####### NVM COMMANDS ########

// check version
node -v || node --version

// list locally installed versions of node
nvm ls

// list remove available versions of node
nvm ls-remote

// install specific version of node
nvm install 18.16.1

// set default version of node
nvm alias default 18.16.1

// switch version of node
nvm use 20.5.1

// install latest LTS version of node (Long Term Support)
nvm install --lts

// install latest stable version of node
nvm install stable
